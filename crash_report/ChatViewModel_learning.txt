   549	        if (screenLearningJob?.isActive == true) {
   550	            screenLearningJob?.cancel()
   551	            screenLearningJob = null
   552	            lastLearnedScreenText = null
   553	            _uiState.update { it.copy(isScreenLearningActive = false) }
   554	
   555	            appendMessage(
   556	                ChatMessage(
   557	                    text = "⏸️ أوقفت وقت التعلم.",
   558	                    isFromUser = false
   559	                )
   560	            )
   561	            return
   562	        }
   563	
   564	        appendMessage(
   565	            ChatMessage(
   566	                text = "🧠 بدأ وقت التعلم. سأتابع الشاشة وأحفظ المعلومات الجديدة محليًا، ولن أحفظ الأسرار أو مفاتيح API.",
   567	                isFromUser = false
   568	            )
   569	        )
   570	
   571	        _uiState.update { it.copy(isScreenLearningActive = true) }
   572	        screenLearningJob = viewModelScope.launch(Dispatchers.IO) {
   573	            while (true) {
   574	                delay(2500L)
   575	
   576	                val snapshot = EveAccessibilityService.instance
   577	                    ?.getLatestScreenSnapshot()
   578	                    ?: continue
   579	
   580	                val safeText = SmartScreenProcessor.cleanAndFormat(
   581	                    snapshot.visibleText
   582	                )
   583	
   584	                if (!SmartScreenProcessor.isWorthAnalyzing(safeText)) {
   585	                    continue
   586	                }
   587	
   588	                if (SmartScreenProcessor.containsSensitiveData(safeText)) {
   589	                    continue
   590	                }
   591	
   592	                if (safeText == lastLearnedScreenText) {
   593	                    continue
   594	                }
   595	
   596	                val now = System.currentTimeMillis()
   597	
   598	                if (now - lastScreenAnalysisAt < 8_000L) {
   599	                    continue
   600	                }
   601	
   602	                lastLearnedScreenText = safeText
   603	                lastScreenAnalysisAt = now
   604	
   605	                val screenPrompt = """
   606	                    حلّل محتوى الشاشة التالي بهدف التعلم طويل المدى.
   607	
   608	                    التطبيق: ${snapshot.appName}

   611	                    المحتوى:
   612	                    $safeText
   613	
   614	                    أعد النتيجة بهذه الصيغة فقط:
   615	                    مفيد: نعم أو لا
   616	                    العنوان: عنوان قصير للمعلومة
   617	                    المعلومة: شرح عربي واضح ومختصر

   623	                    - لا تحفظ كلمات المرور أو مفاتيح API أو الرموز السرية أو البيانات الشخصية.
   624	                    - تجاهل كلمات مثل رجوع، إلغاء، مشاركة، فتح، إغلاق.

   626	                    - إذا لم توجد معلومة قابلة للتعلم، أعد مفيد: لا فقط.
   627	                """.trimIndent()
   628	
   629	                val memory = try {
   630	                    val analysis = withTimeout(25_000L) {
   631	                        qwenService.generateResponse(screenPrompt)
   632	                    }
   633	
   634	                    SmartScreenProcessor.parseScreenMemory(analysis)
   635	                } catch (e: Exception) {
   636	                    android.util.Log.e(
   637	                        "ChatViewModel",
   638	                        "فشل تحليل الشاشة للتعلم",
   639	                        e
   640	                    )
   641	                    null
   642	                }
   643	
   644	                if (memory != null) {
   645	                    val memoryKey = "screen:${snapshot.packageName}:${memory.title}"
   646	                        .lowercase()
   647	                        .replace(Regex("\\s+"), "_")
   648	                        .take(220)
   649	
   650	                    val structuredAnswer = buildString {
   651	                        append("العنوان: ")
   652	                        append(memory.title)
   653	                        append("\nالتطبيق: ")
   654	                        append(snapshot.appName)
   655	                        append("\nالموضوع: ")
   656	                        append(memory.topic)
   657	                        append("\nالمعلومة: ")
   658	                        append(memory.information)
   659	                    }
   660	
   661	                    knowledgeRepo.saveOrUpdate(
   662	                        question = memoryKey,
   663	                        answer = structuredAnswer,
   664	                        topic = memory.topic,
   665	                        keywords = listOf(
   666	                            snapshot.appName,
   667	                            snapshot.packageName,
   668	                            memory.keywords
   669	                        ).joinToString(","),
   670	                        source = "screen_learning"
   671	                    )
   672	                }
   673	            }
   674	        }
   675	    }

   683

   685	                if (learnedContext.isNotBlank()) {

   689	                }
   690	
   691	            val selectedProvider = appPrefs.getLLMProvider()
   692	
   693	            if (
   694	                selectedProvider == "qwen" &&
   695	                qwenService.isModelPresent()

   719	                                isFromUser = false
